function doGet() { return HtmlService.createHtmlOutput("RSVP endpoint is active."); } function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const headers = ["timestamp", "name", "persons", "kids", "service", "cantata"]; const values = sheet.getDataRange().getValues(); if (!values.length) { sheet.appendRow(headers); } else { const firstRow = values[0]; let needsHeader = false; if (firstRow.length < headers.length) { needsHeader = true; } else { for (let i = 0; i < headers.length; i++) { if (firstRow[i] !== headers[i]) { needsHeader = true; break; } } } if (needsHeader) { sheet.insertRowBefore(1); sheet.getRange(1, 1, 1, headers.length).setValues([headers]); } } const data = e.parameter; const row = [ new Date().toISOString(), data.name || "", data.persons || "", data.kids || "", data.service || "", data.cantata || "" ]; sheet.appendRow(row); return ContentService.createTextOutput("OK"); }
